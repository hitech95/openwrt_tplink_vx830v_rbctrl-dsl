cmake_minimum_required(VERSION 3.16)
project(rbctl_voip_module LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
add_compile_options(-Wall -Wextra -Wshadow)

find_path(BARESIP_INCLUDE_DIR baresip.h PATH_SUFFIXES baresip)
find_path(RE_INCLUDE_DIR re.h PATH_SUFFIXES re)
find_path(VOICE_INCLUDE_DIR rbctl_voip.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../../crates/rbctl_voip_ffi/include")
find_library(VOICE_ARCHIVE NAMES librbctl_voip_ffi.a)
foreach(dep BARESIP_INCLUDE_DIR RE_INCLUDE_DIR VOICE_INCLUDE_DIR VOICE_ARCHIVE)
  if(NOT ${dep})
    message(FATAL_ERROR "Set ${dep} to the corresponding target dependency (see README.md)")
  endif()
endforeach()

option(RBCTL_HARNESS "Allow selection of the inherited Unix datagram fixture" OFF)
option(HAVE_THREADS "Match libre's C11 threads configuration" ON)
add_library(rbctl_voip MODULE module.c board.c attach.c call.c audio.c digitmap.c panel.c)
if(HAVE_THREADS)
  target_compile_definitions(rbctl_voip PRIVATE HAVE_THREADS)
endif()
if(RBCTL_HARNESS)
  target_compile_definitions(rbctl_voip PRIVATE RBCTL_HARNESS)
endif()
set_target_properties(rbctl_voip PROPERTIES PREFIX "")
target_compile_features(rbctl_voip PRIVATE c_std_11)
target_include_directories(rbctl_voip PRIVATE "${BARESIP_INCLUDE_DIR}" "${RE_INCLUDE_DIR}" "${VOICE_INCLUDE_DIR}")
find_package(Threads REQUIRED)
target_link_libraries(rbctl_voip PRIVATE "${VOICE_ARCHIVE}" Threads::Threads ${CMAKE_DL_LIBS} m)

include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(test_board tests/board.c board.c)
  target_include_directories(test_board PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${VOICE_INCLUDE_DIR}")
  target_link_libraries(test_board PRIVATE "${VOICE_ARCHIVE}" Threads::Threads ${CMAKE_DL_LIBS} m)
  add_test(NAME board_lifecycle COMMAND test_board)
  find_program(BARESIP_EXE baresip)
  find_path(BARESIP_MODULE_DIR account.so PATH_SUFFIXES lib/baresip/modules)
  if(RBCTL_HARNESS AND BARESIP_EXE AND BARESIP_MODULE_DIR)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME baresip_registration_smoke COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/registration.py" "${BARESIP_EXE}"
      "${BARESIP_MODULE_DIR}" "$<TARGET_FILE:rbctl_voip>")
    set_tests_properties(baresip_registration_smoke PROPERTIES TIMEOUT 15)
    if(EXISTS "${BARESIP_MODULE_DIR}/mwi.so")
      add_test(NAME baresip_mwi_registration COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/mwi.py" "${BARESIP_EXE}" "${BARESIP_MODULE_DIR}" "$<TARGET_FILE:rbctl_voip>")
      set_tests_properties(baresip_mwi_registration PROPERTIES TIMEOUT 20)
    endif()
    foreach(scenario identity failures early)
      add_test(NAME baresip_signalling_${scenario} COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/signalling.py" "${BARESIP_EXE}"
        "${BARESIP_MODULE_DIR}" "$<TARGET_FILE:rbctl_voip>" "${scenario}")
      set_tests_properties(baresip_signalling_${scenario} PROPERTIES TIMEOUT 30)
    endforeach()
    set(call_cases caller callee board-clock reject-tone lost-ack send-failure dual waiting policy timers regblock solo shared shared-isolation hunt shortmap longmap postdisc fax)
    if(EXISTS "${BARESIP_MODULE_DIR}/srtp.so")
      list(APPEND call_cases srtp)
    endif()
    if(EXISTS "${BARESIP_MODULE_DIR}/g722.so" AND EXISTS "${BARESIP_MODULE_DIR}/auresamp.so")
      list(APPEND call_cases g722)
    endif()
    foreach(side IN LISTS call_cases)
      add_test(NAME baresip_call_${side}_hangup COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/calls.py" "${BARESIP_EXE}"
        "${BARESIP_MODULE_DIR}" "$<TARGET_FILE:rbctl_voip>" ${side})
      set_tests_properties(baresip_call_${side}_hangup PROPERTIES TIMEOUT 30)
    endforeach()
    foreach(scenario running missing-state missing-start missing-locale reject-default)
      add_test(NAME baresip_attach_${scenario} COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/registration.py" "${BARESIP_EXE}"
        "${BARESIP_MODULE_DIR}" "$<TARGET_FILE:rbctl_voip>" "${scenario}")
      set_tests_properties(baresip_attach_${scenario} PROPERTIES TIMEOUT 15)
    endforeach()
  else()
    message(STATUS "Registration smoke requires RBCTL_HARNESS=ON and a host baresip 3.16.0 installation")
  endif()
endif()
